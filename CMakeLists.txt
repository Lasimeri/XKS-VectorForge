cmake_minimum_required(VERSION 3.20)
project(xks_vectorforge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(xksv_core src/codec.cpp src/layout.cpp)
set_target_properties(xksv_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(xksv_core PUBLIC include)
add_library(xksv_native SHARED src/c_api.cpp)
target_link_libraries(xksv_native PRIVATE xksv_core)
target_include_directories(xksv_native PRIVATE include)
target_compile_options(xksv_core PRIVATE -Wall -Wextra -Wpedantic)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
add_executable(xksv-native src/inspect.cpp src/compile.cpp src/verify.cpp)
target_include_directories(xksv-native PRIVATE third_party)
target_link_libraries(xksv-native PRIVATE xksv_native OpenSSL::Crypto Threads::Threads)
target_compile_options(xksv-native PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(xksv_tests tests/test_core.cpp)
  target_link_libraries(xksv_tests PRIVATE xksv_core)
  add_test(NAME core COMMAND xksv_tests)
  add_test(NAME core_scalar COMMAND xksv_tests)
  set_tests_properties(core_scalar PROPERTIES ENVIRONMENT "XKSV_FORCE_SCALAR=1")
  add_executable(xksv_inspect_tests tests/test_inspect.cpp)
  target_include_directories(xksv_inspect_tests PRIVATE third_party)
  add_test(NAME native_inspect COMMAND xksv_inspect_tests $<TARGET_FILE:xksv-native>)
endif()
option(XKSV_LEGACY_PYTHON_TESTS "Enable tests for the transitional Python prototype" OFF)
if(BUILD_TESTING AND XKSV_LEGACY_PYTHON_TESTS)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME checkpoint COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_checkpoint.py)
  set_tests_properties(checkpoint PROPERTIES ENVIRONMENT "XKSV_LIBRARY=$<TARGET_FILE:xksv_native>")
endif()
option(XKSV_ENABLE_CUDA "Build experimental CUDA block decoder" OFF)
if(XKSV_ENABLE_CUDA)
  enable_language(CUDA)
  add_library(xksv_cuda SHARED cuda/decode.cu cuda/parallel.cu)
  set_target_properties(xksv_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()
option(XKSV_ENABLE_LLAMA_BRIDGE "Build the experimental Qwen3.5 llama.cpp FUSE bridge" OFF)
set(XKSV_LLAMA_ROOT "" CACHE PATH "llama.cpp source directory")
set(XKSV_LLAMA_BUILD_DIR "${XKSV_LLAMA_ROOT}/build" CACHE PATH "llama.cpp build directory")
if(XKSV_ENABLE_LLAMA_BRIDGE)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FUSE3 REQUIRED IMPORTED_TARGET fuse3)
  find_path(XKSV_GGML_INCLUDE gguf.h HINTS "${XKSV_LLAMA_ROOT}/ggml/include" NO_DEFAULT_PATH REQUIRED)
  find_library(XKSV_GGML_BASE NAMES ggml-base HINTS "${XKSV_LLAMA_BUILD_DIR}/bin" "${XKSV_LLAMA_BUILD_DIR}/ggml/src" NO_DEFAULT_PATH REQUIRED)
  add_executable(xksv-llama-bridge src/llama_bridge.cpp)
  target_include_directories(xksv-llama-bridge PRIVATE third_party "${XKSV_GGML_INCLUDE}")
  target_link_libraries(xksv-llama-bridge PRIVATE xksv_native OpenSSL::Crypto "${XKSV_GGML_BASE}" PkgConfig::FUSE3)
endif()
set(XKSV_TEST_CUDA_PYTHON "" CACHE FILEPATH "Python with PyTorch for optional CUDA tests")
if(BUILD_TESTING AND XKSV_LEGACY_PYTHON_TESTS AND XKSV_ENABLE_CUDA AND XKSV_TEST_CUDA_PYTHON)
  add_test(NAME cuda_kernels COMMAND ${XKSV_TEST_CUDA_PYTHON} ${CMAKE_SOURCE_DIR}/tests/test_cuda.py)
  add_test(NAME streaming_inference COMMAND ${XKSV_TEST_CUDA_PYTHON} ${CMAKE_SOURCE_DIR}/tests/test_stream.py)
  add_test(NAME autotune COMMAND ${XKSV_TEST_CUDA_PYTHON} ${CMAKE_SOURCE_DIR}/tests/test_autotune.py)
  set_tests_properties(autotune PROPERTIES ENVIRONMENT "XKSV_LIBRARY=$<TARGET_FILE:xksv_native>;XKSV_CUDA_LIBRARY=$<TARGET_FILE:xksv_cuda>")
  set_tests_properties(streaming_inference PROPERTIES ENVIRONMENT "XKSV_LIBRARY=$<TARGET_FILE:xksv_native>")
  set_tests_properties(cuda_kernels PROPERTIES ENVIRONMENT "XKSV_LIBRARY=$<TARGET_FILE:xksv_native>;XKSV_CUDA_LIBRARY=$<TARGET_FILE:xksv_cuda>")
endif()
